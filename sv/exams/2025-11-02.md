/*@
logic integer count_row(bool *row, integer c, integer idx) = 
    idx >= c ? 0;

logic integer count matrix(bool **matrix, integer r, integer c, integer idx) =

*/

/*@
    requires count_matrix(lights, n, m, 0) == 0;
    ensures count_matrix(lights, n, m, 0) == 0;
*/
void lightrow(bool **lights, int i) {
// turn on
    /*@
    loop invariant count_matrix(lights, n, m, 0) = j;
    */
  for (int j = 0; j < m; j++) {
    lights[i][j] = 1;
  }

// turn off
/*@
    loop invariant count_matrix(lights, n, m, 0) = m-j;
    */

  for (int j = 0; j < m; j++) {
    lights[i][j] = 0;
  }
}

void lightshow(bool **lights) {
  for (int i = 0; i < n; i++) {
    lightrow(lights, i);
  }
}

/*@
  requires n >= 0;
  requires (\forall integer j; (0 <= j < n) ==> (b <= items[j] <= r));
  ensures (0 <= low <= mid <= n);
  ensures (\forall integer j; (0 <= j < low) ==> items[j] == b);
  ensures (\forall integer j; (low <= j < mid) ==> items[j] == w);
  ensures (\forall integer j; (mid <= j < n) ==> items[j] == r);
@*/
void FrenchNationalFlagSort(int items[], int n) {
  low = 0;
  mid = 0;
  int high = n - 1;


/*@
loop invariant 0 <= mid <= high;
loop invariant 0 <= low <= high;
loop invariant 0 <= high <= n-1;
loop assigns mid, low, high, items[low], items[mid]
loop variant high - mid;
*/
  while (mid <= high) {
    if (items[mid] == b) {
      // blue case
      if (low != mid) {
        swap(items, low, mid);
      }
      low = low + 1;
      mid = mid + 1;
    }
    else {
      if (items[mid] == w) {
        // white case
        mid = mid + 1;
      } else {
        // red case
        if (mid != high) {
          swap(items, mid, high);
        }
        high = high - 1;
      }
    }
  }
}

void count count_visitors(){
  bool p = true;
  bool q = *;
  while(*){ // neither predicate says anything about i or nr_of_visitors
    if(q){
      p = p ? true : *;
      q = *
    } 
    else {
      p = !p ? false: *;
      q = *;
    }
    // p doesn’t involve i, so it stays the same; q now refers to the next visitor, which is unknown
    p = p;
    q = *;
  }
}
