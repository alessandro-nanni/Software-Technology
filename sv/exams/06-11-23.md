/*@
  requires votes_number >= 0;
  requires \valid(votes + (0..votes_number-1));
  requires \valid(votes_per_position + (0..positions-1));
  requires \forall integer j; 0 <= j < votes_number ==> 0 <= votes[j] < positions;
  requires \forall integer i; 0 <= i < positions ==> votes_per_position[i] == 0;
  requires \separated(votes + (0..votes_number-1), votes_per_position + (0..positions-1));

  assigns votes_per_position[0..positions-1];

  ensures \result == votes_per_position;
  ensures \forall integer i; 0 <= i < positions ==> votes_per_position[i] >= 0;
  ensures \forall integer i; 0 <= i < positions ==>
            votes_per_position[i] == \numof(0, votes_number-1, \lambda integer j; votes[j] == i);
*/
int* count(int votes[], int votes_number, int votes_per_position[], int positions) {
  /*@
    loop invariant 0<= i <= votes_number;
    loop invarant \forall integer k; 0 <= k <= i ==> votes_per_position[k] >= 0
    loop invariant \forall integer k; 0 <= k <= i ==>
            votes_per_position[i] == \numof(0, votes_number-1, \lambda integer j; votes[k] == i);
    loop assigns i, votes_per_position[0..positions-1];
    loop variant votes_number - i;
  */
  for (int i = 0; i < votes_number; i++) {
    votes_per_position[votes[i]]++;
  }
  return votes_per_position;
}
