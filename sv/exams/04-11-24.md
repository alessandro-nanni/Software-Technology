int needs_maintenance(track_info tracks[], int track_number, int maintenance_threshold, track_info *output) {
  int j = 0;
    /*@
        loop invariant 0 <= i <= track_number;
        loop invariant 0 <= j <= i;
        loop invariant \forall integer k; 0 <= k <= j ==> output[k].sensitive || output[k].days_since_maintenance >= maintenance_threshold;
        loop assigns i, j, output[0..j]
        loop variant track_number - i;
    */
  for (int i = 0; i < track_number; i++) {
    if (tracks[i].days_since_maintenance >= maintenance_threshold || tracks[i].sensitive) {
      output[j] = tracks[i];
      j++;
    }
  }
  return j;
}
